Google Sheet Two-Way Sync & Email Webhook

Connect your Southern Electronics Google Sheet in 2 minutes

Leave blank to run with built-in local persistence. Paste your deployed Google Apps Script Web App URL to sync rows directly with Google Sheets and send automated Gmail alerts.

Google Apps Script Code (Extensions → Apps Script) 
function doGet(e) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Renewals");
  const rows = sheet.getDataRange().getValues();
  const headers = rows.shift();
  const records = rows.map(r => Object.fromEntries(headers.map((h, i) => [h, r[i]])));
  return ContentService.createTextOutput(JSON.stringify({ status: "ok", records }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  const payload = JSON.parse(e.postData.contents);
  if (payload.action === "sendEmail") {
    MailApp.sendEmail({
      to: payload.email,
      subject: "SEONICS Renewal Alert: " + payload.category + " (" + payload.refNumber + ")",
      body: payload.message
    });
  }
  return ContentService.createTextOutput(JSON.stringify({ status: "ok" }))
    .setMimeType(ContentService.MimeType.JSON);
}
